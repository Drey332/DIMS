1) “Fire Intelligence” incident schema (v0.1)

These fields are consistent with how your API & storage are shaped today and map cleanly to future ML features. Copy as-is into @shared/schema (or keep local).

// src/fire-intel/schema.ts
import { z } from "zod";

export const FireIncidentZ = z.object({
  id: z.string(),                                // "PIPER-ALPHA-1988"
  name: z.string(),
  dateUtc: z.string(),                           // ISO 8601
  industry: z.enum(["offshore_oil_gas","mining","onshore_oil_gas","refining","other"]),
  location: z.string(),
  operationPhase: z.enum(["production","drilling","completion","workover","maintenance","construction","unknown"]),
  initiatingEvent: z.string(),                   // free text
  ignitionSource: z.string().optional(),
  fuel: z.array(z.string()),                     // e.g. ["condensate","gas","oil"]
  detection: z.array(z.string()),                // e.g. ["gas detectors","operator observed"]
  protectionSystems: z.object({
    delugeStatus: z.enum(["operational","degraded","failed","not_installed"]).optional(),
    eStopIsolation: z.enum(["effective","delayed","failed","not_attempted"]).optional(),
    blowoutPreventer: z.enum(["effective","degraded","failed","not_applicable"]).optional(),
    fireWater: z.enum(["available","insufficient","failed","not_applicable"]).optional(),
    alarmsPA: z.enum(["effective","degraded","failed","not_applicable"]).optional()
  }).partial(),
  humanFactors: z.array(z.string()),             // “permit-to-work lapse”, “misread test”, “training gap”
  barriersFailed: z.array(z.string()),           // maps to BowTie/Swiss-cheese barriers
  fatalities: z.number().min(0),
  injuries: z.number().min(0).optional(),
  assetLossUSD_2025: z.number().optional(),
  releaseVolume: z.string().optional(),          // human-friendly (“> 4MM bbl oil spilled”)
  timeline: z.array(z.object({ t: z.string(), event: z.string() })), // key moments
  officialFindings: z.array(z.string()),         // short, evidence-based bullets
  sources: z.array(z.object({ title: z.string(), url: z.string() })),
  lessons: z.array(z.string()),                  // normalized lessons we’ll use for rules/ML
  tags: z.array(z.string())                      // quick filters (“fire”, “explosion”, “BOP”, “deluge”)
});
export type FireIncident = z.infer<typeof FireIncidentZ>;

b) A tiny persistence+indexer you can run once (or on boot)

// src/fire-intel/ingest.ts
import fs from "fs/promises";
import path from "path";
import { FireIncidentZ, type FireIncident } from "./schema";
import { storage } from "@/server/storage"; // your storage adapter
import { createEmbedding } from "@/server/ai/embeddings"; // your existing embedding util (OpenAI or local)

function toKnowledgeText(i: FireIncident) {
  const head = `${i.name} (${i.dateUtc}) — ${i.location}`;
  const facts = [
    `Industry: ${i.industry}, Phase: ${i.operationPhase}`,
    `Initiating event: ${i.initiatingEvent}`,
    `Ignition: ${i.ignitionSource ?? "unknown"}, Fuel: ${i.fuel.join(", ")}`,
    `Protection: ${JSON.stringify(i.protectionSystems)}`,
    `Fatalities: ${i.fatalities}${i.injuries ? `, Injuries: ${i.injuries}` : ""}`,
    `Barriers failed: ${i.barriersFailed.join("; ")}`,
    `Lessons: ${i.lessons.join("; ")}`
  ].join("\n");
  return `${head}\n${facts}\nTimeline:\n${i.timeline.map(x=>`- ${x.t}: ${x.event}`).join("\n")}\nSources:\n${i.sources.map(s=>`- ${s.title}: ${s.url}`).join("\n")}`;
}

export async function ingestFireIncidents(seedFile = path.join(process.cwd(), "data", "fire-incidents.seed.json")) {
  const raw = await fs.readFile(seedFile, "utf-8");
  const arr = JSON.parse(raw) as unknown[];
  const valid: FireIncident[] = [];

  for (const rec of arr) {
    const parsed = FireIncidentZ.safeParse(rec);
    if (!parsed.success) {
      console.error("Invalid incident record:", parsed.error.flatten());
      continue;
    }
    valid.push(parsed.data);
  }

  for (const i of valid) {
    // 1) persist in your DB (idempotent upsert)
    await storage.upsertKnowledgeItem({
      domain: "fire_incident",
      externalId: i.id,
      title: i.name,
      payload: i,
      tags: i.tags
    });

    // 2) vectorize for fast retrieval in ERP/Advisor prompts
    const text = toKnowledgeText(i);
    const emb = await createEmbedding(text);
    await storage.upsertVector({
      externalId: i.id,
      namespace: "fire_incident:v0",
      embedding: emb,
      metadata: { phase: i.operationPhase, fatalities: i.fatalities, tags: i.tags }
    });
  }

  console.log(`Ingested ${valid.length} fire incidents → knowledge + vectors`);
}

If you don’t have storage.upsertKnowledgeItem and upsertVector yet, mirror your existing createFileUpload style — same idea: a table for canonical JSON (jsonb), and a vectors table (pgvector) keyed by externalId.

4) Instant “safety thresholds” (rules that work even if APIs fail)

Put this tiny engine in src/fire-intel/rules.ts. It will auto-flag dangerous states using what we learned from the seeds:

export type Telemetry = {
  gasPpm?: number;                 // fixed detectors
  delugeReady?: boolean;           // proof-tested
  eStopHealthy?: boolean;
  bopMode?: "closed"|"open"|"unknown"; // if drilling/completion
  negPressureTest?: "pass"|"fail"|"ambiguous"|"not_applicable";
  flareStatus?: "available"|"down";
  simultaneousOps?: boolean;
  hotWorkActive?: boolean;
  permitIsolationVerified?: boolean;
};

export function evaluateFireRisk(phase: "production"|"drilling"|"completion"|"maintenance", t: Telemetry) {
  const findings: string[] = [];
  let score = 0;

  // Lessons baked into rules (Piper Alpha)
  if (phase === "production") {
    if (t.delugeReady === false) { findings.push("Deluge not ready during production (Piper Alpha lesson)"); score += 4; }
    if (t.hotWorkActive && !t.permitIsolationVerified) { findings.push("Hot work without verified isolation (permit-to-work failure)"); score += 5; }
    if (t.gasPpm && t.gasPpm > 20) { findings.push("Elevated gas reading >20 ppm — treat as leak until proven safe"); score += 3; }
  }

  // Lessons (Macondo)
  if (phase === "completion") {
    if (t.negPressureTest === "ambiguous" || t.negPressureTest === "fail") {
      findings.push("Negative pressure test not clearly passing — DO NOT displace, shut in well.");
      score += 6;
    }
    if (t.bopMode !== "closed") { findings.push("BOP not closed during anomaly — shut in immediately"); score += 4; }
  }

  // Generic
  if (t.eStopHealthy === false) { findings.push("E-stop / isolation degraded"); score += 3; }
  if (t.flareStatus === "down") { findings.push("Flare not available — no safe disposal path"); score += 2; }
  if (t.simultaneousOps) { findings.push("SIMOPS active — raise supervision level"); score += 1; }

  const level = score >= 8 ? "CRITICAL" : score >= 4 ? "HIGH" : score >= 2 ? "MEDIUM" : "LOW";
  return { level, score, findings };
}

Wire this into your /api/ai/protocol-guidance (or a new /api/fire/risk) so the UI can show a red/yellow/green risk badge even when external feeds are down.
