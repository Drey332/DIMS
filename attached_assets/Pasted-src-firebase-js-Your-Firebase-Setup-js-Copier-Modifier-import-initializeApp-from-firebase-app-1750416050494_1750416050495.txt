import { useState } from "react";
import { auth, googleProvider } from "../firebase";
import {
  signInWithPopup,
  signInWithEmailAndPassword,
  createUserWithEmailAndPassword,
  signOut
} from "firebase/auth";

export default function Login() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [user, setUser] = useState(null);
  const [error, setError] = useState("");
  const [mode, setMode] = useState("login"); // or "register"

  // Listen for auth changes (will also catch Google login)
  auth.onAuthStateChanged((user) => {
    setUser(user);
  });

  // Email/password login
  const handleLogin = async (e) => {
    e.preventDefault();
    setError("");
    try {
      const userCred = await signInWithEmailAndPassword(auth, email, password);
      setUser(userCred.user);
    } catch (err) {
      setError(err.message);
    }
  };

  // Email/password registration
  const handleRegister = async (e) => {
    e.preventDefault();
    setError("");
    try {
      const userCred = await createUserWithEmailAndPassword(auth, email, password);
      setUser(userCred.user);
    } catch (err) {
      setError(err.message);
    }
  };

  // Google login
  const handleGoogleLogin = async () => {
    setError("");
    try {
      const result = await signInWithPopup(auth, googleProvider);
      setUser(result.user);
    } catch (err) {
      setError(err.message);
    }
  };

  // Sign out
  const handleLogout = async () => {
    await signOut(auth);
    setUser(null);
  };

  if (user) {
    return (
      <div style={{ maxWidth: 400, margin: "2rem auto", padding: 24, border: "1px solid #ddd", borderRadius: 8 }}>
        <h2>Welcome, {user.displayName || user.email}</h2>
        <p><b>Email:</b> {user.email}</p>
        {user.photoURL && <img src={user.photoURL} alt="Avatar" style={{ width: 60, borderRadius: "50%" }} />}
        <button onClick={handleLogout} style={{ marginTop: 16, padding: "8px 24px" }}>Log Out</button>
      </div>
    );
  }

  return (
    <div style={{ maxWidth: 400, margin: "2rem auto", padding: 24, border: "1px solid #ddd", borderRadius: 8 }}>
      <h2>{mode === "login" ? "Sign In" : "Register"}</h2>
      {error && <div style={{ color: "red", marginBottom: 8 }}>{error}</div>}
      <form onSubmit={mode === "login" ? handleLogin : handleRegister}>
        <input
          style={{ width: "100%", margin: "8px 0", padding: 8 }}
          type="email"
          placeholder="Email"
          value={email}
          onChange={e => setEmail(e.target.value)}
          required
        />
        <input
          style={{ width: "100%", margin: "8px 0", padding: 8 }}
          type="password"
          placeholder="Password"
          value={password}
          onChange={e => setPassword(e.target.value)}
          required
        />
        <button
          type="submit"
          style={{
            width: "100%",
            background: "#1a73e8",
            color: "white",
            padding: "10px 0",
            border: "none",
            borderRadius: 4,
            marginBottom: 8,
            marginTop: 4
          }}>
          {mode === "login" ? "Sign In" : "Register"}
        </button>
      </form>
      <div style={{ textAlign: "center", margin: "8px 0" }}>OR</div>
      <button
        onClick={handleGoogleLogin}
        style={{
          width: "100%",
          background: "#fff",
          color: "#444",
          padding: "10px 0",
          border: "1px solid #ddd",
          borderRadius: 4
        }}>
        Continue with Google
      </button>
      <div style={{ textAlign: "center", marginTop: 12 }}>
        {mode === "login" ? (
          <span>Don't have an account? <button style={{ color: "#1a73e8", background: "none", border: "none" }} onClick={() => setMode("register")}>Register</button></span>
        ) : (
          <span>Already have an account? <button style={{ color: "#1a73e8", background: "none", border: "none" }} onClick={() => setMode("login")}>Sign In</button></span>
        )}
      </div>
    </div>
  );
}
3. How It Works:
You can register/sign-in with email/password (using only Firebase, no backend!).

Or, you can click “Continue with Google” and it’ll log you in with your Google account.

All errors (like popup blocked, wrong password, etc.) are shown at the top.

If you are logged in, you’ll see your user info and a Log Out button.

No fetch to /api/auth, no backend needed—all handled by Firebase.
