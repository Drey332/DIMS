// src/fire-intel/geo.ts
export function toSeason(dateIso: string, lat: number) {
  // Simple: Northern vs Southern hemisphere
  const m = new Date(dateIso).getUTCMonth() + 1;
  const north = lat >= 0;
  const seasonMapNorth: Record<number,string> = {1:"winter",2:"winter",3:"spring",4:"spring",5:"spring",6:"summer",7:"summer",8:"summer",9:"fall",10:"fall",11:"fall",12:"winter"};
  const seasonMapSouth: Record<number,string> = {1:"summer",2:"summer",3:"fall",4:"fall",5:"fall",6:"winter",7:"winter",8:"winter",9:"spring",10:"spring",11:"spring",12:"summer"};
  return (north ? seasonMapNorth[m] : seasonMapSouth[m]) as "winter"|"spring"|"summer"|"fall";
}

export function latBand(lat: number) {
  const a = Math.abs(lat);
  if (a < 23.5) return "tropical";
  if (a < 45)   return "subtropical";
  if (a < 66.5) return "midlat";
  return "polar";
}

// very coarse world-ocean basins (tune later)
export function basin(lat: number, lon: number) {
  if (lat > 0 && lon > -100 && lon < 20) return "North_Atlantic";
  if (lat > 0 && lon >= 20 && lon < 180) return "North_Indian_Pacific";
  if (lat < 0 && lon > -80 && lon < 40) return "South_Atlantic_Indian";
  return "Pacific_Other";
}

2) Add a matching endpoint that blends semantic similarity + geo/season/env boosts

// src/server/routes/incidents-match.ts
import type { Request, Response } from "express";
import { storage } from "@/server/storage";
import { lookupLocationIntel } from "@shared/environment/locationIntel";
import { getEnvContext } from "@/server/env-intel"; // you have this

type MatchReq = {
  query?: string;                 // user question (optional)
  location?: string;              // "Forcados, Nigeria"
  lat?: number; lon?: number;
  whenUtc?: string;               // ISO; fallback = now
  phase?: "production"|"drilling"|"completion"|"maintenance"|"construction"|"unknown";
};

function seasonFrom(lat: number, iso: string) {
  const m = new Date(iso).getUTCMonth()+1;
  const north = lat >= 0;
  const sn = {1:"winter",2:"winter",3:"spring",4:"spring",5:"spring",6:"summer",7:"summer",8:"summer",9:"fall",10:"fall",11:"fall",12:"winter"} as const;
  const ss = {1:"summer",2:"summer",3:"fall",4:"fall",5:"fall",6:"winter",7:"winter",8:"winter",9:"spring",10:"spring",11:"spring",12:"summer"} as const;
  return (north ? sn[m] : ss[m]);
}

export async function registerIncidentMatchRoute(app: import("express").Express) {
  app.get("/api/incidents/match", async (req: Request, res: Response) => {
    try {
      const q: MatchReq = {
        query: typeof req.query.q === "string" ? req.query.q : undefined,
        location: typeof req.query.location === "string" ? req.query.location : undefined,
        lat: req.query.lat ? Number(req.query.lat) : undefined,
        lon: req.query.lon ? Number(req.query.lon) : undefined,
        whenUtc: typeof req.query.whenUtc === "string" ? req.query.whenUtc : undefined,
        phase: typeof req.query.phase === "string" ? req.query.phase as any : undefined,
      };

      const locIntel = lookupLocationIntel(q.location);
      const lat = q.lat ?? locIntel?.latitude ?? 0;
      const lon = q.lon ?? locIntel?.longitude ?? 0;
      const when = q.whenUtc ?? new Date().toISOString();
      const season = seasonFrom(lat, when);

      // Optional: pull live env snapshot for subtle boosts (wind, waves)
      // keep timeouts small and fail-soft
      let env: any = null;
      try { env = await getEnvContext(lat, lon); } catch {}

      const vecResults = await storage.searchVectors({
        namespace: "fire_incident:v0",
        query: q.query ?? "offshore fire explosion lessons barriers",
        k: 8
      });

      // Re-rank with **transparent boosts**:
      const boosted = vecResults.map((r: any) => {
        let score = r.score; // base vector similarity (higher is better)
        if (q.phase && r.metadata?.phase === q.phase) score += 0.15;
        if (r.metadata?.latBand && Math.abs(Math.abs(lat) - Math.abs(r.metadata.latitude||0)) < 10) score += 0.1;
        if (r.metadata?.basin && r.metadata.basin.includes("Atlantic") && Math.abs(lon) < 100) score += 0.05;
        if (r.metadata?.season === season) score += 0.05;
        // wind-driven spread: if current wind > 12 m/s, prefer incidents with big escalation
        if (env?.wind?.currentSpeed && env.wind.currentSpeed > 12 && r.metadata?.tags?.includes("explosion")) score += 0.05;
        return { ...r, boostedScore: score };
      })
      .sort((a: any, b: any) => b.boostedScore - a.boostedScore)
      .slice(0, 2);

      // Return minimal, auditable payload
      const items = await Promise.all(boosted.map(async (r: any) => {
        const item = await storage.getKnowledgeByExternalId("fire_incident", r.externalId);
        return {
          id: r.externalId,
          title: item?.title ?? r.externalId,
          location: item?.payload?.location,
          dateUtc: item?.payload?.dateUtc,
          operationPhase: item?.payload?.operationPhase,
          lessons: item?.payload?.lessons ?? [],
          officialFindings: item?.payload?.officialFindings ?? [],
          sources: item?.payload?.sources ?? [],
          score: r.boostedScore,
        };
      }));

      res.json({
        context: { lat, lon, season, whenUtc: when, phase: q.phase ?? "unknown" },
        matches: items
      });
    } catch (e) {
      console.error(e);
      res.status(500).json({ message: "Failed to compute incident matches" });
    }
  });
}
