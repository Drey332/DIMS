    console.error('API responded with error:', errorText);
    throw new Error(errorText);
  }
  const result = await resp.json();
  console.log('Team member added:', result);
} catch (err) {
  console.error('Fetch failed:', err);
  alert('Error: ' + err.message);
}

Step 5: Production-Grade API Utility for Team Add
Here’s a reusable utility to ensure you are sending the correct request:

js
Copier
Modifier
async function addTeamMember({ firstName, lastName, email, number, rank, token }) {
  const payload = {
    name: `${firstName} ${lastName}`,
    email,
    number,
    rank, // gold, silver, bronze
  };

  console.log('Payload:', payload);

  try {
    const resp = await fetch('/api/team', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        ...(token && { 'Authorization': 'Bearer ' + token }),
      },
      body: JSON.stringify(payload),
    });
    if (!resp.ok) {
      const errorText = await resp.text();
      throw new Error(errorText || 'Unknown server error');
    }
    return await resp.json();
  } catch (err) {
    console.error('Add team member error:', err);
    throw err;
  }
}
Summary of Troubleshooting Steps
Check every fetch call: it must be /api/team, not /api/team-members.

Log everything: what are you sending, what’s the endpoint, and what does the backend receive?

Check browser DevTools Network tab: what’s the exact failing endpoint and method?

Backend logging: make sure your POST /api/team route is logging incoming requests.

Check for method errors: You must use POST, not GET, for adding.

Restart everything after changes (frontend + backend).
