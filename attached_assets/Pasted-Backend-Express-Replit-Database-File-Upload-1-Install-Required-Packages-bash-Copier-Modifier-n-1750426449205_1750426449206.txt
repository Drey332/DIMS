const express = require('express');
const multer = require('multer');
const { Database } = require("replitdb");
const cors = require('cors');
const fs = require('fs');
const path = require('path');

const app = express();
app.use(cors());
app.use(express.json());
app.use('/uploads', express.static('uploads')); // Serve photos

const db = new Database();

// Multer for local file uploads (to /uploads)
const upload = multer({ dest: './uploads/' });

// List all assets
app.get('/api/assets', async (req, res) => {
  const assets = (await db.get("assets")) || [];
  res.json(assets);
});

// Add new asset
app.post('/api/assets', async (req, res) => {
  const { name, location } = req.body;
  if (!name || !location) return res.status(400).json({ error: "Missing fields" });
  const assets = (await db.get("assets")) || [];
  const asset = {
    id: Date.now().toString(),
    name,
    location,
    createdAt: Date.now(),
    verifications: [],
  };
  assets.push(asset);
  await db.set("assets", assets);
  res.json(asset);
});

// Verify asset (with photo, location, comment, etc.)
app.post('/api/assets/:id/verify', upload.single('photo'), async (req, res) => {
  const { id } = req.params;
  const { comment, lat, lng, verifiedBy } = req.body;
  const photoUrl = req.file ? `/uploads/${req.file.filename}` : null;
  let assets = (await db.get("assets")) || [];
  const idx = assets.findIndex(a => a.id === id);
  if (idx === -1) return res.status(404).json({ error: "Asset not found" });
  const verification = {
    photoUrl,
    timestamp: Date.now(),
    location: { lat, lng },
    comment,
    verifiedBy,
  };
  assets[idx].verifications = assets[idx].verifications || [];
  assets[idx].verifications.push(verification);
  await db.set("assets", assets);
  res.json(assets[idx]);
});

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => console.log('HydroSafe Asset Backend (Replit DB) running on', PORT));
🟨 Frontend: React Asset Verification (No changes needed)
You can use the previous React code!
Just make sure your API endpoints are pointed to /api/assets and /api/assets/:id/verify.

Assets are stored as a list at key "assets" in Replit DB.

Each asset has an array of verifications, each with photoUrl, timestamp, comment, location, and verifiedBy.

Photos are stored in /uploads folder in your Replit workspace.

🟦 Advanced CTO Tips
To limit DB size, periodically prune old assets or photos from /uploads.
