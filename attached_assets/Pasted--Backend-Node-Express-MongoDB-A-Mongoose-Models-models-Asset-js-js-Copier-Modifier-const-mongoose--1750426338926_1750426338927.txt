const VerificationSchema = new mongoose.Schema({
  photoUrl: String,
  timestamp: { type: Date, default: Date.now },
  location: { lat: Number, lng: Number },
  comment: String,
  verifiedBy: String, // User ID or name
});
const AssetSchema = new mongoose.Schema({
  name: String,
  location: String,
  createdAt: { type: Date, default: Date.now },
  verifications: [VerificationSchema],
});
module.exports = mongoose.model('Asset', AssetSchema);

const express = require('express');
const router = express.Router();
const multer = require('multer');
const Asset = require('../models/Asset');

// Use multer for photo upload (local, for simplicity; swap for S3/GCS for prod)
const upload = multer({ dest: './uploads/' });

// Create asset
router.post('/', async (req, res) => {
  const { name, location } = req.body;
  if (!name || !location) return res.status(400).json({ error: "Missing fields" });
  const asset = new Asset({ name, location });
  await asset.save();
  res.json(asset);
});

// Get all assets
router.get('/', async (req, res) => {
  const assets = await Asset.find().sort({ createdAt: -1 });
  res.json(assets);
});

// Verify asset: upload photo + data
router.post('/:id/verify', upload.single('photo'), async (req, res) => {
  const { id } = req.params;
  const { lat, lng, comment, verifiedBy } = req.body;
  // Store file path (swap for cloud storage URL in production)
  const photoUrl = req.file ? req.file.path : null;
  const asset = await Asset.findById(id);
  if (!asset) return res.status(404).json({ error: "Asset not found" });
  asset.verifications.push({
    photoUrl,
    timestamp: new Date(),
    location: { lat, lng },
    comment,
    verifiedBy,
  });
  await asset.save();
  res.json(asset);
});
module.exports = router;

import { useState, useEffect } from "react";

export default function AssetVerification() {
  const [assets, setAssets] = useState([]);
  const [form, setForm] = useState({ name: "", location: "" });
  const [verifyModal, setVerifyModal] = useState(null); // { asset, open }
  const [verifyData, setVerifyData] = useState({ photo: null, comment: "", lat: "", lng: "", verifiedBy: "" });

  // Fetch asset list
  useEffect(() => {
    fetch('/api/assets').then(r => r.json()).then(setAssets);
  }, []);

  // Add new asset
  async function handleAdd(e) {
    e.preventDefault();
    const resp = await fetch('/api/assets', {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify(form)
    });
    if (!resp.ok) return alert("Failed to add asset");
    const asset = await resp.json();
    setAssets(a => [asset, ...a]);
    setForm({ name: "", location: "" });
  }

  // Open verify modal for asset
  function openVerify(asset) {
    setVerifyData({ photo: null, comment: "", lat: "", lng: "", verifiedBy: "" });
    setVerifyModal(asset);
  }

  // Handle verify (with photo upload)
  async function handleVerify(e) {
    e.preventDefault();
    const fd = new FormData();
    fd.append('photo', verifyData.photo);
    fd.append('comment', verifyData.comment);
    fd.append('lat', verifyData.lat);
    fd.append('lng', verifyData.lng);
    fd.append('verifiedBy', verifyData.verifiedBy);
    const resp = await fetch(`/api/assets/${verifyModal._id}/verify`, { method: "POST", body: fd });
    if (!resp.ok) return alert("Failed to verify asset");
    const updated = await resp.json();
    setAssets(a => a.map(ax => ax._id === updated._id ? updated : ax));
    setVerifyModal(null);
  }

  return (
    <div className="max-w-2xl mx-auto p-6">
      <h2 className="text-2xl font-bold mb-4">Asset Verification</h2>
      <form className="flex gap-2 mb-6" onSubmit={handleAdd}>
        <input className="border p-2 rounded flex-1" placeholder="Asset Name" value={form.name} required onChange={e => setForm(f => ({ ...f, name: e.target.value }))} />
        <input className="border p-2 rounded flex-1" placeholder="Location" value={form.location} required onChange={e => setForm(f => ({ ...f, location: e.target.value }))} />
        <button className="bg-blue-600 text-white px-4 rounded" type="submit">Add Asset</button>
      </form>
      <ul className="space-y-4">
        {assets.map(asset => (
          <li key={asset._id} className="p-4 bg-white rounded-xl shadow flex flex-col gap-2">
            <div className="flex justify-between items-center">
              <div>
                <span className="font-bold">{asset.name}</span>
                <span className="ml-4 text-gray-500">{asset.location}</span>
              </div>
              <button onClick={() => openVerify(asset)} className="bg-green-600 text-white px-4 py-1 rounded">Verify</button>
            </div>
            {asset.verifications && asset.verifications.length > 0 && (
              <div className="border-t pt-2 mt-2">
                <span className="font-bold">Last Verification:</span>
                {asset.verifications.map((v, idx) => (
                  <div key={idx} className="mb-2">
                    <div><b>By:</b> {v.verifiedBy || "Unknown"} <b>@</b> {new Date(v.timestamp).toLocaleString()}</div>
                    <div><b>Location:</b> {v.location.lat},{v.location.lng}</div>
                    <div><b>Comment:</b> {v.comment}</div>
                    {v.photoUrl && <img src={`/${v.photoUrl.replace(/^\.?\//, '')}`} alt="Verification" className="w-32 mt-1" />}
                  </div>
                )).slice(-1)}
              </div>
            )}
          </li>
        ))}
      </ul>

      {/* Modal for verify */}
      {verifyModal && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
          <form className="bg-white p-8 rounded-xl shadow-xl w-[350px] flex flex-col gap-3" onSubmit={handleVerify}>
            <h3 className="font-bold text-lg mb-2">Verify: {verifyModal.name}</h3>
            <input required className="border p-2 rounded" placeholder="Your Name" value={verifyData.verifiedBy} onChange={e => setVerifyData(d => ({ ...d, verifiedBy: e.target.value }))} />
            <input required className="border p-2 rounded" type="number" step="any" placeholder="Latitude" value={verifyData.lat} onChange={e => setVerifyData(d => ({ ...d, lat: e.target.value }))} />
            <input required className="border p-2 rounded" type="number" step="any" placeholder="Longitude" value={verifyData.lng} onChange={e => setVerifyData(d => ({ ...d, lng: e.target.value }))} />
            <input required className="border p-2 rounded" type="file" accept="image/*" capture="environment" onChange={e => setVerifyData(d => ({ ...d, photo: e.target.files[0] }))} />
            <textarea required className="border p-2 rounded" placeholder="Briefly describe what you saw" value={verifyData.comment} onChange={e => setVerifyData(d => ({ ...d, comment: e.target.value }))} />
            <button className="bg-green-600 text-white px-4 py-2 rounded mt-2">Submit Verification</button>
            <button type="button" className="text-gray-600 mt-1" onClick={() => setVerifyModal(null)}>Cancel</button>
          </form>
        </div>
      )}
    </div>
  );
}
🟩 How It Works
Add Asset: Clean form, must provide name & location.

List updates instantly after add.

Verify: Modal opens, you upload photo, input geolocation, name, comment.

Photo is captured or uploaded, location is entered (can enhance with GPS APIs), timestamp is auto-stored.

Comment field for describing what was checked.

Verifications: Show as a history/log under each asset, with time, user, location, photo, and comment.

All data is in MongoDB; images are saved locally (/uploads/). For cloud, swap multer storage engine.

🟦 Top CTO Enhancements & Tips
Use GPS APIs to autofill location fields.

Cloud Storage: Swap Multer to upload to AWS S3, Google Cloud Storage, or Replit Blob if needed.

WebSockets: Use Socket.io for real-time updates across users.

Role-Based Access: Only allow Gold/Silver users to verify or add assets if needed.

Audit Log: Keep full verification trail for compliance.

AI image check: Optionally auto-analyze photo quality or content for extra safety.
