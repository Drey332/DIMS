const express = require('express');
const router = express.Router();
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');
const User = require('../models/User');

// Use Replit secrets
const GOLD_CODE = process.env.GOLD_CODE || '000';
const JWT_SECRET = process.env.JWT_SECRET || 'replace_this_in_replit_secrets';

router.post('/register', async (req, res) => {
  const { email, password, name, goldCode } = req.body;
  const hash = await bcrypt.hash(password, 10);
  const isGold = goldCode === GOLD_CODE;
  const user = new User({ email, password: hash, name, isGold });
  await user.save();
  const token = jwt.sign({ userId: user._id, isGold }, JWT_SECRET, { expiresIn: "7d" });
  res.json({ token, user: { email, name, isGold } });
});

router.post('/login', async (req, res) => {
  const { email, password } = req.body;
  const user = await User.findOne({ email });
  if (!user || !await bcrypt.compare(password, user.password))
    return res.status(400).json({ error: "Invalid credentials" });
  const token = jwt.sign({ userId: user._id, isGold: user.isGold }, JWT_SECRET, { expiresIn: "7d" });
  res.json({ token, user: { email: user.email, name: user.name, isGold: user.isGold } });
});

router.post('/gold-unlock', async (req, res) => {
  const { userId, code } = req.body;
  if (code !== GOLD_CODE) return res.status(400).json({ error: "Invalid Gold Command code" });
  await User.findByIdAndUpdate(userId, { isGold: true });
  res.json({ success: true });
});

module.exports = router;
4. routes/team.js
js
Copier
Modifier
const express = require('express');
const router = express.Router();
const TeamMember = require('../models/TeamMember');
const jwt = require('jsonwebtoken');

const JWT_SECRET = process.env.JWT_SECRET || 'replace_this_in_replit_secrets';

function requireGold(req, res, next) {
  const auth = req.headers.authorization;
  if (!auth) return res.status(401).json({ error: "Missing token" });
  try {
    const { isGold } = jwt.verify(auth.split(' ')[1], JWT_SECRET);
    if (!isGold) return res.status(403).json({ error: "Gold Command only" });
    next();
  } catch {
    return res.status(401).json({ error: "Invalid or expired token" });
  }
}

// Public: Get all members
router.get('/', async (req, res) => {
  const team = await TeamMember.find();
  team.sort((a, b) => {
    const order = { gold: 0, silver: 1, bronze: 2 };
    return order[a.rank] - order[b.rank];
  });
  res.json(team);
});

// Add member (Gold only)
router.post('/', requireGold, async (req, res) => {
  const { name, email, number, rank } = req.body;
  const member = new TeamMember({ name, email, number, rank });
  await member.save();
  res.json(member);
});

// Edit member (Gold only)
router.put('/:id', requireGold, async (req, res) => {
  const member = await TeamMember.findByIdAndUpdate(req.params.id, req.body, { new: true });
  res.json(member);
});

// Delete member (Gold only)
router.delete('/:id', requireGold, async (req, res) => {
  await TeamMember.findByIdAndDelete(req.params.id);
  res.json({ success: true });
});

module.exports = router;

const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());

// Connect MongoDB
const MONGO_URI = process.env.MONGO_URI || 'mongodb://localhost:27017/hydrosafe';
mongoose.connect(MONGO_URI, { useNewUrlParser: true, useUnifiedTopology: true });

// API routes
app.use('/api/auth', require('./routes/auth'));
app.use('/api/team', require('./routes/team'));

// Start server
const PORT = process.env.PORT || 5000;
app.listen(PORT, () => console.log('Server running on port', PORT));

**7. How to add Replit secrets (env vars)
Go to your Replit workspace.

Click the padlock icon (“Secrets”) in the left sidebar.

Add these:

JWT_SECRET = a-very-strong-secret-key

GOLD_CODE = 000

MONGO_URI = your MongoDB URI (use MongoDB Atlas for free cloud hosting, or keep mongodb://localhost:27017/hydrosafe for local dev)

8. Deploy/Run
npm install

Click “Run” in Replit or use node server.js
