// Example middleware to restrict editing to Gold role only
function requireGoldRole(req, res, next) {
  if (req.user.role === "Gold") {
    next();
  } else {
    res.status(403).json({ error: "Only Gold can edit projects" });
  }
}

// Routes
app.post('/api/projects', requireGoldRole, createProject);     // Create project
app.put('/api/projects/:id', requireGoldRole, editProject);    // Edit project

3. React Frontend Component: Project Tabs
jsx
Copier
Modifier
// Props: userRole, projects (array), activeProjectId, onSelect, onCreate (Gold only)

function ProjectTabs({ userRole, projects, activeProjectId, onSelect, onCreate }) {
  return (
    <div className="flex items-center bg-gray-50 p-2 border-b">
      <div className="flex space-x-2 flex-1">
        {projects.map(project => (
          <button
            key={project.id}
            className={`px-4 py-2 rounded-t-lg ${activeProjectId === project.id
              ? 'bg-blue-500 text-white font-bold'
              : 'bg-gray-200 text-gray-700'
            }`}
            onClick={() => onSelect(project.id)}
          >
            {project.name}
          </button>
        ))}
      </div>
      {userRole === 'Gold' && (
        <button
          className="ml-4 bg-green-500 text-white px-4 py-2 rounded"
          onClick={onCreate}
        >
          + New Project
        </button>
      )}
    </div>
  );
}

How It Works:
Gold users see a "+ New Project" button and can click any project to edit settings, assign roles, or upload docs.

Silver/Bronze see all projects they are assigned to as tabs; they can switch between them and see project details and dashboard, but cannot edit project data.

All incident management, assets, and communication are always tied to the currently active project.
